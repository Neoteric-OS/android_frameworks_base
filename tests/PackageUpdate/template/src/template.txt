    @Override
    public void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.main);
        mPkgName = getPackageName();
        try {
            mPkgInfo = getPackageManager().getPackageInfo(mPkgName, PackageManager.GET_ACTIVITIES);
        } catch (NameNotFoundException e) {
            e.printStackTrace();
            throw new RuntimeException(e);
        }

        appendPackageInfoToBuffer();

        Log.d(TAG_INFO, mInfoBuffer.toString());
        if (mDiagBuffer.length() > 0) {
            Log.d(TAG_DIAG, mDiagBuffer.toString());
        }

        TextView view = (TextView) findViewById(R.id.textView);
        view.setText(mInfoBuffer.toString() + "\n\n" + mDiagBuffer.toString());
    }

    @Override

        super.onResume();
    }

    private void appendPackageInfoToBuffer() {
        mInfoBuffer.append(String.format("Package :%%s\n",
                mPkgName));
        mInfoBuffer.append(String.format("VersionName :%%s\n",
                mPkgInfo.versionName));
        mInfoBuffer.append(String.format("VersionCode :%%s\n",
                mPkgInfo.versionCode));
        String resver = getString(R.integer.versionCode);
        if (! resver.equals(Integer.toString(mPkgInfo.versionCode))){
            mDiagBuffer.append("* Something wrong with Resource Version.\n");
        }
        mInfoBuffer.append(String.format("ResourceVersion:%%s\n", resver));
        mInfoBuffer.append(String.format("SourceDir   :%%s\n",
                mPkgInfo.applicationInfo.sourceDir));
        mInfoBuffer.append(String.format("publicSourceDir   :%%s\n",
                mPkgInfo.applicationInfo.publicSourceDir));
        mInfoBuffer.append(String.format("DataDir     :%%s\n",
                mPkgInfo.applicationInfo.dataDir));
        mInfoBuffer.append(String.format("NativeLibDir:%%s\n",
                mPkgInfo.applicationInfo.nativeLibraryDir));

        appendApplicationFlagsInfo();

        mInfoBuffer.append(String.format("splitSourceDirs :%%s\n",
                Arrays.toString(mPkgInfo.applicationInfo.splitSourceDirs)));
        mInfoBuffer.append(String.format("splitPublicSourceDirs :%%s\n",
                Arrays.toString(mPkgInfo.applicationInfo.splitPublicSourceDirs)));
        mInfoBuffer.append(String.format("sharedLibraryFiles :%%s\n",
                Arrays.toString(mPkgInfo.applicationInfo.sharedLibraryFiles)));
        mInfoBuffer.append(String.format("scanSourceDir   :%%s\n",
                getFieldValue(mPkgInfo.applicationInfo, "scanSourceDir")));
        mInfoBuffer.append(String.format("scanPublicSourceDir   :%%s\n",
                getFieldValue(mPkgInfo.applicationInfo, "scanPublicSourceDir")));
        mInfoBuffer.append(String.format("secondaryNativeLibraryDir   :%%s\n",
                getFieldValue(mPkgInfo.applicationInfo, "secondaryNativeLibraryDir")));
        // mInfoBuffer.append(String.format("nativeLibraryRootDir   :%%s\n",
        //        getFieldValue(mPkgInfo.applicationInfo, "nativeLibraryRootDir")));

        try {
            mInfoBuffer.append(String.format("NativeLibVersion:%%s\n", getLibraryVersion()));
        } catch (UnsatisfiedLinkError e) {
            mInfoBuffer.append("NativeLibVersion: Error\n");
            mDiagBuffer.append("* UnsatisfiedLinkError occured.\n");
        }

        updateSharedPreference();
    }

    private void appendApplicationFlagsInfo() {
        ApplicationFlags applicationFlags = new ApplicationFlagsForO();
        int flags = mPkgInfo.applicationInfo.flags;
        mInfoBuffer.append(String.format("flags:0x%%08X\n", flags));

        applicationFlags.appendFlagDetailInfo(mInfoBuffer, flags);

        int privateFlags = (int) getFieldValue(mPkgInfo.applicationInfo, "privateFlags");
        mInfoBuffer.append(String.format("privateFlags:0x%%08X\n", privateFlags));

        applicationFlags.appendPrivateFlagDetailInfo(mInfoBuffer, privateFlags);
    }

    private void updateSharedPreference() {
        SharedPreferences pref = getSharedPreferences(SP_NAME_EMULATED_USERDATA,
                Activity.MODE_PRIVATE);

        int ver = pref.getInt(SP_KEY_DATA_VERSION_CURRENT, -1);
        String history = pref.getString(SP_KEY_DATA_VERSION_HISTORY, "");

        if (ver == -1) {
            mDiagBuffer.append("* EMPTY UserData.\n");
        } else if (ver > mPkgInfo.versionCode) {
            this.mDiagBuffer.append(String.format(
                    "* UserData version is %%s but app version is %%s\n", ver, mPkgInfo.versionCode));
        }
        if (history != "") {
            history += ",";
        }
        history += mPkgInfo.versionCode;

        SharedPreferences.Editor editor = pref.edit();
        editor.putString(SP_KEY_DATA_VERSION_HISTORY, history);
        editor.putInt(SP_KEY_DATA_VERSION_CURRENT, mPkgInfo.versionCode);
        editor.commit();

        mInfoBuffer.append(String.format("UserDataVersion :%%s\n", ver + " => " + mPkgInfo.versionCode));
        mInfoBuffer.append(String.format("UserDataHistory :%%s\n", history));
    }

    abstract static class ApplicationFlags {
        void appendFlagDetailInfo(StringBuilder buffer, int flag) {
            appendAnyFlagsDetailInfo(buffer, flag, getTargetFlags());
        }
        void appendPrivateFlagDetailInfo(StringBuilder buffer, int privateFlag) {
            appendAnyFlagsDetailInfo(buffer, privateFlag, getTargetPrivateFlags());
        }
        protected abstract String[] getTargetFlags();
        protected abstract String[] getTargetPrivateFlags();
        protected void appendAnyFlagsDetailInfo(StringBuilder buffer, int flags, String[] targetFields) {
            for (String targetField : targetFields) {
                boolean isFlagSet = ((int)getFieldValue(this, targetField) & flags) != 0;
                buffer.append(String.format("    %%s : %%s\n", targetField, isFlagSet));
            }
        }
    }

    private static Object getFieldValue(Object instance, String field) {
        try {
            java.lang.reflect.Field f = instance.getClass().getDeclaredField(field);
            f.setAccessible(true);
            return f.get(instance);
        } catch (ReflectiveOperationException e) {
            Log.e(TAG_INFO, field, e);
            return "<failed to get value>";
        }
    }
